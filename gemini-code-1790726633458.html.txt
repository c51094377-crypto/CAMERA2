        function selectRole(role) {

        function handlePinInput(input, index) {
            const val = input.value.replace(/[^0-9]/g, '');
            input.value = val;
            const inputs = document.querySelectorAll('.pin-input');
            if (val && index < 3) inputs[index + 1].focus();
            checkPinComplete();
        }

        function handlePinKeyDown(event, index) {
            const inputs = document.querySelectorAll('.pin-input');
            if (event.key === 'Backspace' && !inputs[index].value && index > 0) {
                inputs[index - 1].focus();
            }
        }

        function checkPinComplete() {
            const inputs = document.querySelectorAll('.pin-input');
            let code = '';
            inputs.forEach(i => code += i.value);
            const btn = document.getElementById('connect-btn');
            if (code.length === 4) {
                btn.removeAttribute('disabled');
                currentCode = code;
            } else {
                btn.setAttribute('disabled', 'true');
            }
        }

        // モニター側ロックモード用の動体自動追尾枠描画ループ
        let trackingInterval = null;
        let prevTrackingFrameData = null;

        function startTrackingLoop() {
            if (trackingInterval) clearInterval(trackingInterval);
            const remoteVideo = document.getElementById('remote-video');
            const trackingCanvas = document.getElementById('tracking-canvas');
            const ctx = trackingCanvas.getContext('2d');
            const sampleCanvas = document.createElement('canvas');
            const sCtx = sampleCanvas.getContext('2d');

            trackingInterval = setInterval(() => {
                if (currentMonitorMode !== 'lock') return;
                if (!remoteVideo.videoWidth || !remoteVideo.videoHeight) return;

                if (trackingCanvas.width !== remoteVideo.videoWidth || trackingCanvas.height !== remoteVideo.videoHeight) {
                    trackingCanvas.width = remoteVideo.videoWidth;
                    trackingCanvas.height = remoteVideo.videoHeight;
                }

                sampleCanvas.width = 160;
                sampleCanvas.height = 90;
                sCtx.drawImage(remoteVideo, 0, 0, 160, 90);
                const imgData = sCtx.getImageData(0, 0, 160, 90);
                const pixels = imgData.data;

                ctx.clearRect(0, 0, trackingCanvas.width, trackingCanvas.height);

                if (prevTrackingFrameData) {
                    let minX = 160, minY = 90, maxX = 0, maxY = 0;
                    let diffPixelsCount = 0;

                    for (let y = 0; y < 90; y++) {
                        for (let x = 0; x < 160; x++) {
                            const i = (y * 160 + x) * 4;
                            const rDiff = Math.abs(pixels[i] - prevTrackingFrameData[i]);
                            const gDiff = Math.abs(pixels[i+1] - prevTrackingFrameData[i+1]);
                            const bDiff = Math.abs(pixels[i+2] - prevTrackingFrameData[i+2]);
                            if ((rDiff + gDiff + bDiff) / 3 > 30) {
                                diffPixelsCount++;
                                if (x < minX) minX = x;
                                if (y < minY) minY = y;
                                if (x > maxX) maxX = x;
                                if (y > maxY) maxY = y;
                            }
                        }
                    }

                    // 動きを検知した場合に四角い追尾枠を描画
                    if (diffPixelsCount > 80 && maxX > minX && maxY > minY) {
                        const scaleX = trackingCanvas.width / 160;
                        const scaleY = trackingCanvas.height / 90;

                        const boxX = minX * scaleX;
                        const boxY = minY * scaleY;
                        const boxW = (maxX - minX) * scaleX;
                        const boxH = (maxY - minY) * scaleY;

                        ctx.strokeStyle = '#10b981'; // エメラルドグリーン
                        ctx.lineWidth = 4;
                        ctx.strokeRect(boxX - 10, boxY - 10, boxW + 20, boxH + 20);

                        // 四隅のカッコ装飾
                        ctx.fillStyle = '#10b981';
                        ctx.font = 'bold 14px monospace';
                        ctx.fillText('⚡ LOCK TRACKING', boxX - 10, boxY - 18);
                    }
                }
                prevTrackingFrameData = pixels;
            }, 150);
        }

        function manualPlayVideo() {
            if (currentRole === 'monitor') {

                if (remoteVideo && remoteVideo.srcObject) {

                        document.getElementById('status-text').textContent = '接続成功 (リアルタイム監視中)';
                        startTrackingLoop();
                    }).catch(e => console.error("[Monitor] 手動play()失敗:", e));
                }
            }
        }

        // --- カメラ側 (iPad A) ---
        async function connectWithCode() {
            const btn = document.getElementById('connect-btn');
            btn.textContent = '接続中...';
            btn.setAttribute('disabled', 'true');
            document.getElementById('status-text').textContent = 'カメラを起動中...';

            try {
                localStream = await navigator.mediaDevices.getUserMedia({ 
                    video: { facingMode: 'environment', width: { ideal: 1280 }, height: { ideal: 720 } }, 
                    audio: true 
                });
                console.log("[Camera] getUserMedia成功:", localStream);

                document.getElementById('view-camera-setup').classList.add('hidden');
                document.getElementById('view-active-session').classList.remove('hidden');
                document.getElementById('camera-controls').classList.remove('hidden');

                const localVideo = document.getElementById('local-video');
                localVideo.srcObject = localStream;
                localVideo.classList.remove('hidden');
                document.getElementById('video-placeholder').classList.add('hidden');
                document.getElementById('video-title').textContent = 'カメラ側プレビュー (iPad A)';
                document.getElementById('connection-dot').classList.replace('bg-emerald-500', 'bg-blue-500');
                document.getElementById('status-text').textContent = 'モニターを探索中...';

                // 同一ブラウザ内でも競合しないように設定
                peer = new Peer(undefined, { debug: 2 });

                peer.on('open', (id) => {
                    console.log("[Camera] Peerオープン成功 ID:", id);
                    
                    // モニター側のPeerIDを特定（部分一致検索のため、部屋名ベースで接続試行）
                    const targetRoomPrefix = 'ipad-cam-room-' + currentCode;
                    document.getElementById('status-text').textContent = '接続を確立しています...';
                    
                    // 同一ブラウザ環境でモニター側IDを正確に紐付けるため直接接続
                    // 相手の正確なPeerIDが不明な場合を考慮し、接続コードから推測または自動探索
                    setTimeout(() => {
                        // モニター側へ接続試行（同一ブラウザ対応のためプレフィックスを結合）
                        connectToMonitorRoom(targetRoomPrefix, localStream);
                    }, 500);
                });

                peer.on('error', (err) => {
                    console.error("[Camera] Peerエラー:", err);
                    alert('モニターが見つかりません。4桁コードが正しいか確認してください。');
                    location.reload();
                });

            } catch (err) {
                console.error("[Camera] カメラ初期化エラー:", err);
                alert('カメラ・マイクの権限が許可されていないか、利用できません。');
                btn.textContent = '接続';
                btn.removeAttribute('disabled');
            }
        }

        // 同一ブラウザ・別デバイス両対応の接続関数
        function connectToMonitorRoom(roomPrefix, stream) {
            // PeerJSでは完全一致IDが必要なため、同一ブラウザ内テスト用に対応IDを補正
            // ※別端末の場合はそのまま完全一致するか、PeerJSサーバーの仕様に合わせます
            // 通常運用（別端末）の標準ID構造
            let targetPeerId = roomPrefix;

            console.log("[Camera] モニターへ接続試行:", targetPeerId);

            // データ通信チャネルの確立（画像同期用）
            dataConn = peer.connect(targetPeerId);
            setupDataConnectionListeners();

            // ライブ映像・音声の発信
            activeCall = peer.call(targetPeerId, stream);

            if (activeCall) {
                activeCall.on('open', () => {
                    console.log("[Camera] call接続確立成功");
                    document.getElementById('status-text').textContent = '接続成功 (カメラ稼働中)';
                    document.getElementById('connection-dot').classList.replace('bg-blue-500', 'bg-emerald-500');
                    initMotionDetection();
                });

                activeCall.on('close', () => {
                    console.log("[Camera] 接続が切断されました");
                    alert('接続が切断されました。');
                    location.reload();
                });
            } else {
                alert('モニターとの接続に失敗しました。コードを再確認してください。');
                location.reload();
            }
        }

        // --- データチャネル設定 ---
        function setupDataConnectionListeners() {
            if (!dataConn) return;
            dataConn.on('data', (data) => {
                console.log("[DataChannel] データ受信:", data.type);
                if (data.type === 'SNAPSHOT') {
                    appendSnapshotToGallery(data.url, data.reason, data.time, false);
                }
            });
        }

        // --- 動体検知・自動撮影機能（1秒間隔に短縮） ---
        let motionInterval = null;
        let lastImageData = null;
        let lastCapturedTime = 0;

        function toggleMotionDetection(checkbox) {
            const statusEl = document.getElementById('motion-status');
            if (checkbox.checked) {
                statusEl.textContent = '動体検知: 有効 (1秒間隔)';
                statusEl.classList.replace('bg-slate-700', 'bg-emerald-900/60');
                statusEl.classList.replace('text-slate-400', 'text-emerald-300');
                startMotionDetectionLoop();
            } else {
                statusEl.textContent = '動体検知: 停止中';
                statusEl.classList.replace('bg-emerald-900/60', 'bg-slate-700');
                statusEl.classList.replace('text-emerald-300', 'text-slate-400');
                stopMotionDetectionLoop();
            }
        }

        function initMotionDetection() {
            const toggle = document.getElementById('motion-toggle');
            toggle.checked = true;
            toggleMotionDetection(toggle);
        }

        function startMotionDetectionLoop() {
            const video = document.getElementById('local-video');
            const canvas = document.getElementById('motion-canvas');
            const ctx = canvas.getContext('2d');

            if (motionInterval) clearInterval(motionInterval);

            // 3秒(3000ms)から「1秒(1000ms)」間隔に変更
            motionInterval = setInterval(() => {
                if (!video.videoWidth || !video.videoHeight) return;
                
                canvas.width = 160;
                canvas.height = 90;
                ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
                
                const imgData = ctx.getImageData(0, 0, canvas.width, canvas.height);
                const pixels = imgData.data;

                if (lastImageData) {
                    let diffCount = 0;
                    for (let i = 0; i < pixels.length; i += 16) {
                        const rDiff = Math.abs(pixels[i] - lastImageData[i]);
                        const gDiff = Math.abs(pixels[i+1] - lastImageData[i+1]);
                        const bDiff = Math.abs(pixels[i+2] - lastImageData[i+2]);
                        if ((rDiff + gDiff + bDiff) / 3 > 30) diffCount++;
                    }

                    const now = Date.now();
                    // 連続撮影のクールダウンも1秒間隔に合わせて調整（1000ms）
                    if (diffCount > 100 && (now - lastCapturedTime > 1000)) {
                        lastCapturedTime = now;
                        triggerCapture('動体検知自動撮影');
                    }
                }
                lastImageData = pixels;
            }, 300); // 処理チェック自体は高頻度で行い、判定を1秒間隔のクールダウンで制御
        }

        function stopMotionDetectionLoop() {
            if (motionInterval) {
                clearInterval(motionInterval);
                motionInterval = null;
            }
        }

        function triggerManualCapture() {
            if (currentRole !== 'camera') return;
            triggerCapture('手動撮影');
        }

        function triggerCapture(reason) {
            const video = document.getElementById('local-video');
            if (!video || !video.videoWidth) return;

            const canvas = document.createElement('canvas');
            canvas.width = video.videoWidth;
            canvas.height = video.videoHeight;
            const ctx = canvas.getContext('2d');
            ctx.drawImage(video, 0, 0, canvas.width, canvas.height);

            const dataUrl = canvas.toDataURL('image/jpeg', 0.8);
            const timeStr = new Date().toLocaleTimeString();

            appendSnapshotToGallery(dataUrl, reason, timeStr, true);

            if (dataConn && dataConn.open) {
                dataConn.send({
                    type: 'SNAPSHOT',
                    url: dataUrl,
                    reason: reason,
                    time: timeStr
                });
            }
        }

        function appendSnapshotToGallery(dataUrl, reason, timeStr, isLocal) {
            const gallery = document.getElementById('snapshot-gallery');
            if (gallery.querySelector('.text-center')) {
                gallery.innerHTML = '';
            }

            snapshotCount++;
            document.getElementById('snapshot-count').textContent = snapshotCount + '枚';

            const item = document.createElement('div');
            item.className = 'bg-slate-900 rounded-xl overflow-hidden border border-slate-700 shadow flex flex-col animate-fade-in';
            item.innerHTML = `
                <a href="${dataUrl}" target="_blank">
                    <img src="${dataUrl}" class="w-full h-36 object-cover hover:opacity-90 transition">
                </a>
                <div class="p-2.5 flex justify-between items-center text-xs">
                    <span class="font-bold text-amber-400">${reason}</span>
                    <span class="text-slate-400">${timeStr}</span>
                </div>
            `;
            gallery.prepend(item);
        }
    </script>
